# Exact next steps (direct VM / nginx / systemd — NOT Docker)

# 1. VM prerequisites
sudo useradd -r -s /bin/false archflux 2>/dev/null || true
sudo mkdir -p /var/www/html /var/www/archflux-server /etc/archflux
sudo cp deploy/backend.env.example /etc/archflux/backend.env
sudo chmod 600 /etc/archflux/backend.env
# EDIT /etc/archflux/backend.env: JWT_SECRET, DATABASE_URL, DB_USERNAME, DB_PASSWORD, APP_API_TOKEN, APP_COOKIE_SECURE=true
sudo cp deploy/archflux-backend.service /etc/systemd/system/
sudo cp deploy/nginx-default.conf /etc/nginx/sites-available/default
sudo ln -sf /etc/nginx/sites-available/default /etc/nginx/sites-enabled/default
sudo nginx -t && sudo systemctl reload nginx
sudo systemctl daemon-reload
sudo fallocate -l 2G /swapfile || true
sudo chmod 600 /swapfile || true
sudo mkswap /swapfile || true
sudo swapon /swapfile || true
echo '/swapfile none swap sw 0 0' | sudo tee -a /etc/fstab || true
cd server && mvn -q -DskipTests package || true

# 2. Enable backend
sudo systemctl enable --now archflux-backend

# 3. Verify before push
curl -sf http://127.0.0.1:3001/api/health || echo "FAIL health"
curl -sf -o /dev/null -w "%{http_code}" http://localhost/ || echo "FAIL web"

# 4. Push code (triggers .github/workflows/deploy.yml)
git add .
git commit -m "deploy setup"
git push origin main

# 5. Verify after push (wait ~2-3 min)
curl -sf http://127.0.0.1:3001/api/health
curl -sf -o /dev/null -w "%{http_code}" http://localhost/
