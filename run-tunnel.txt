# Deployment commands for this environment (Docker Compose — NOT systemd/n直接)
# Environment: mailtoravi437@cloudshell (container/cloudshell — no systemd)

# 1. Clone / pull repo
mkdir -p ~/archflux
git clone https://github.com/mailtoravi437/archflux.git ~/archflux || (cd ~/archflux && git pull origin main)

# 2. Set .env secrets (copy example, edit real values)
cp ~/archflux/.env.example ~/archflux/.env || true
# Edit with: POSTGRES_PASSWORD, DB_APP_PASSWORD, API_TOKEN, JWT_SECRET, APP_COOKIE_SECURE=true

# 3. Add swap (optional but recommended)
sudo fallocate -l 2G /swapfile 2>/dev/null || true
sudo chmod 600 /swapfile 2>/dev/null || true
sudo mkswap /swapfile 2>/dev/null || true
sudo swapon /swapfile 2>/dev/null || true
echo '/swapfile none swap sw 0 0' | sudo tee -a /etc/fstab 2>/dev/null || true

# 4. Build + deploy (Docker Compose — original project method)
cd ~/archflux
docker compose -f compose.yml up -d --build

# 5. Verify (port 8080 — NOT 3001; NOT systemctl)
sleep 60
docker ps
echo "Health (port 8080):"
curl -sf localhost:8080/api/health || echo "FAIL health"
echo "Ready (port 8080):"
curl -sf localhost:8080/api/ready || echo "FAIL ready"
echo "Web (port 8080):"
curl -sf -o /dev/null -w "%{http_code}" localhost:8080/ || echo "FAIL web"

# 6. Tunnel (optional — same as original run-tunnel.txt)
cloudflared tunnel --url http://localhost:8080 run archflux || true
